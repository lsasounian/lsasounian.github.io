Teste de checkpoint — interrupt_before

@pytest.mark.asyncio
async def test_graph_resumes_after_interruption():
    checkpointer = MemorySaver()

    # Compila com interrupt_before — para ANTES de node_c
    graph = build_test_graph(checkpointer,
        interrupt_before=["node_c"])

    config = {"configurable": {"thread_id": "test-resume-001"}}

    # Primeira execução — para antes de node_c
    await graph.ainvoke({"messages": [], "schema_version": 2}, config=config)

    state = await graph.aget_state(config)
    assert state.next == ("node_c",)   # nó pendente
    assert "node_c" not in execution_log  # ainda não executou

    # Retomada — simula o que o RecoveryConsumer faz
    await graph.ainvoke(None, config=config)  # input=None

    assert "node_c" in execution_log       # agora executou
    state_final = await graph.aget_state(config)
    assert not state_final.next              # estado terminal

Teste de circuit breaker — transições

@pytest.mark.asyncio
async def test_half_open_after_timeout(cb):
    # Abre o circuit
    for _ in range(3):
        with pytest.raises(Exception):
            await cb.call(failing_fn)

    assert cb.state.value == "open"

    # Aguarda recovery_timeout (0.1s no fixture de teste)
    await asyncio.sleep(0.15)

    # Deve permitir UMA chamada de teste (HALF_OPEN)
    result = await cb.call(ok_fn)
    assert result == "ok"

@pytest.mark.asyncio
async def test_closes_after_success_threshold(cb):
    for _ in range(3):
        with pytest.raises(Exception):
            await cb.call(failing_fn)

    await asyncio.sleep(0.15)

    # success_threshold=2 — dois sucessos fecham o circuit
    await cb.call(ok_fn)
    await cb.call(ok_fn)
    assert cb.state.value == "closed"
